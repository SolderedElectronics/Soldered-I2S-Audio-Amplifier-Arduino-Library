79 typedef struct FLACMetadataBlock_t {
80 // METADATA_BLOCK_STREAMINFO
81 uint16_t minblocksize; // The minimum block size (in samples) used in the stream.
82 //----------------------------------------------------------------------------------------
83 // The maximum block size (in samples) used in the stream.
84 uint16_t maxblocksize; // (Minimum blocksize == maximum blocksize) implies a fixed-blocksize stream.
85 //----------------------------------------------------------------------------------------
86 // The minimum frame size (in bytes) used in the stream.
87 uint32_t minframesize; // May be 0 to imply the value is not known.
88 //----------------------------------------------------------------------------------------
89 // The maximum frame size (in bytes) used in the stream.
90 uint32_t maxframesize; // May be 0 to imply the value is not known.
91 //----------------------------------------------------------------------------------------
92 // Sample rate in Hz. Though 20 bits are available,
93 // the maximum sample rate is limited by the structure of frame headers to 655350Hz.
94 uint32_t sampleRate; // Also, a value of 0 is invalid.
95 //----------------------------------------------------------------------------------------
96 // Number of channels FLAC supports from 1 to 8 channels
97 uint8_t numChannels; // 000 : 1 channel .... 111 : 8 channels
98 //----------------------------------------------------------------------------------------
99 // Sample size in bits:
100 // 000 : get from STREAMINFO metadata block
101 // 001 : 8 bits per sample
102 // 010 : 12 bits per sample
103 // 011 : reserved
104 // 100 : 16 bits per sample
105 // 101 : 20 bits per sample
106 // 110 : 24 bits per sample
107 uint8_t bitsPerSample; // 111 : reserved
108 //----------------------------------------------------------------------------------------
109 // Total samples in stream. 'Samples' means inter-channel sample,
110 // i.e. one second of 44.1Khz audio will have 44100 samples regardless of the number
111 uint64_t totalSamples; // of channels. A value of zero here means the number of total samples is unknown.
112 //----------------------------------------------------------------------------------------
113 uint32_t audioDataLength; // is not the filelength, is only the length of the audio datablock in bytes
114 } FLACMetadataBlock_t;

116 typedef struct FLACFrameHeader_t {
117 // 0 : fixed-blocksize stream; frame header encodes the frame number
118 uint8_t blockingStrategy; // 1 : variable-blocksize stream; frame header encodes the sample number
119 //----------------------------------------------------------------------------------------
120 // Block size in inter-channel samples:
121 // 0000 : reserved
122 // 0001 : 192 samples
123 // 0010-0101 : 576 * (2^(n-2)) samples, i.e. 576/1152/2304/4608
124 // 0110 : get 8 bit (blocksize-1) from end of header
125 // 0111 : get 16 bit (blocksize-1) from end of header
126 uint8_t blockSizeCode; // 1000-1111 : 256 * (2^(n-8)) samples, i.e. 256/512/1024/2048/4096/8192/16384/32768
127 //----------------------------------------------------------------------------------------
128 // 0000 : get from STREAMINFO metadata block
129 // 0001 : 88.2kHz
130 // 0010 : 176.4kHz
131 // 0011 : 192kHz
132 // 0100 : 8kHz
133 // 0101 : 16kHz
134 // 0110 : 22.05kHz
135 // 0111 : 24kHz
136 // 1000 : 32kHz
137 // 1001 : 44.1kHz
138 // 1010 : 48kHz
139 // 1011 : 96kHz
140 // 1100 : get 8 bit sample rate (in kHz) from end of header
141 // 1101 : get 16 bit sample rate (in Hz) from end of header
142 // 1110 : get 16 bit sample rate (in tens of Hz) from end of header
143 uint8_t sampleRateCode; // 1111 : invalid, to prevent sync-fooling string of 1s
144 //----------------------------------------------------------------------------------------
145 // Channel assignment
146 // 0000 1 channel: mono
147 // 0001 2 channels: left, right
148 // 0010 3 channels
149 // 0011 4 channels
150 // 0100 5 channels
151 // 0101 6 channels
152 // 0110 7 channels
153 // 0111 8 channels
154 // 1000 : left/side stereo: channel 0 is the left channel, channel 1 is the side(difference) channel
155 // 1001 : right/side stereo: channel 0 is the side(difference) channel, channel 1 is the right channel
156 // 1010 : mid/side stereo: channel 0 is the mid(average) channel, channel 1 is the side(difference) channel
157 uint8_t chanAsgn; // 1011-1111 : reserved
158 //----------------------------------------------------------------------------------------
159 // Sample size in bits:
160 // 000 : get from STREAMINFO metadata block
161 // 001 : 8 bits per sample
162 // 010 : 12 bits per sample
163 // 011 : reserved
164 // 100 : 16 bits per sample
165 // 101 : 20 bits per sample
166 // 110 : 24 bits per sample
167 uint8_t sampleSizeCode; // 111 : reserved
168 //----------------------------------------------------------------------------------------
169 uint32_t totalSamples; // totalSamplesInStream
170 //----------------------------------------------------------------------------------------
171 uint32_t bitrate; // bitrate
172 } FLACFrameHeader_t;
