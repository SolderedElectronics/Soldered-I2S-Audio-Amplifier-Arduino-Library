412 // convert UTF-16 to UTF-8 and stop at zero terminator
413 size_t copy_from_utf16(const uint8_t* src, bool is_big_endian = false) {
414 if (!src) {
415 log_e("arg. is null");
416 return 0;
417 }
418 std::vector<char> out;
419 size_t i = 0;
420
421 // BOM-Handling
422 if (src[i] == 0xFF && src[i + 1] == 0xFE) {
423 is_big_endian = false; // UTF-16LE
424 i += 2;
425 } else if (src[i] == 0xFE && src[i + 1] == 0xFF) {
426 is_big_endian = true; // UTF-16BE
427 i += 2;
428 }
429
430 while (true) {
431 // Prüfe, ob genug Bytes für ein UTF-16-Zeichen vorhanden sind
432 if (i + 1 >= std::numeric_limits<size_t>::max() || (src[i] == 0x00 && src[i + 1] == 0x00)) {
433 break; // Nullterminator oder Ende des Puffers
434 }
435
436 uint16_t ch;
437 if (is_big_endian) {
438 ch = (src[i] << 8) | src[i + 1];
439 } else {
440 ch = (src[i + 1] << 8) | src[i];
441 }
442 i += 2;
443
444 uint32_t codepoint = ch;
445
446 // Prüfe auf Surrogatenpaare
447 if (ch >= 0xD800 && ch <= 0xDBFF) { // High surrogate
448 if ((i + 1 >= std::numeric_limits<size_t>::max()) || (src[i] == 0x00 && src[i + 1] == 0x00)) {
449 log_e("Invalid surrogate pair: missing low surrogate");
450 break;
451 }
452 uint16_t ch2;
453 if (is_big_endian) {
454 ch2 = (src[i] << 8) | src[i + 1];
455 } else {
456 ch2 = (src[i + 1] << 8) | src[i];
457 }
458 if (ch2 < 0xDC00 || ch2 > 0xDFFF) {
459 log_e("Invalid surrogate pair: invalid low surrogate");
460 break;
461 }
462 i += 2;
463 codepoint = 0x10000 + ((ch - 0xD800) << 10) + (ch2 - 0xDC00);
464 } else if (ch >= 0xDC00 && ch <= 0xDFFF) {
465 log_e("Invalid surrogate pair: unexpected low surrogate");
466 break;
467 }
468
469 // UTF-16 → UTF-8
470 if (codepoint < 0x80) {
471 out.push_back(static_cast<char>(codepoint));
472 } else if (codepoint < 0x800) {
473 out.push_back(0xC0 | (codepoint >> 6));
474 out.push_back(0x80 | (codepoint & 0x3F));
475 } else if (codepoint < 0x10000) {
476 out.push_back(0xE0 | (codepoint >> 12));
477 out.push_back(0x80 | ((codepoint >> 6) & 0x3F));
478 out.push_back(0x80 | (codepoint & 0x3F));
479 } else if (codepoint < 0x110000) {
480 out.push_back(0xF0 | (codepoint >> 18));
481 out.push_back(0x80 | ((codepoint >> 12) & 0x3F));
482 out.push_back(0x80 | ((codepoint >> 6) & 0x3F));
483 out.push_back(0x80 | (codepoint & 0x3F));
484 } else {
485 log_e("Invalid codepoint");
486 break;
487 }
488 }
489
490 // Nullterminator hinzufügen
491 out.push_back('\0');
492
493 // Speicher allozieren und kopieren
494 std::size_t bytes = out.size();
495 alloc(bytes);
496 std::memcpy(mem.get(), out.data(), bytes);
497 return i;
498 }

880 va_list args_copy;
881 va_copy(args_copy, args);
882
883 // Ergebnis wird hier reingeschrieben
884 int total_written = 0;
885
886 const char* p = fmt;
887 while (*p) {
888 if (*p == '%') {
889 [[maybe_unused]] const char* start = p++;
890 if (*p == '%') {
891 // Escaped "%%"
892 if (total_written < (int)size) {
893 if (buf) buf[total_written] = '%';
894 }
895 total_written++;
896 p++;
897 continue;
898 }
899
900 // collect complete format tokens now(e.g. "%02d", "%-15.8s" ...)
901 char fmt_token[64];
902 char* f = fmt_token;
903 *f++ = '%';
904 while (*p && !strchr("diuoxXfFeEgGaAcspn%", *p)) { *f++ = *p++; }
905 if (*p) *f++ = *p++;
906 *f = '\0';
907
908 if (fmt_token[std::strlen(fmt_token) - 1] == 's') {
909 // String-Argument abfangen
910 char* s = va_arg(args_copy, char*);
911 if (!s) s = (char*)"(null)";
912 int n = snprintf(buf ? buf + total_written : nullptr, size > (size_t)total_written ? size - total_written : 0, fmt_token, s);
913 if (n < 0) {
914 va_end(args_copy);
915 return n;
916 }
917 total_written += n;
918 } else {
919 // alle anderen Typen → normal weitergeben
920 va_list tmp;
921 va_copy(tmp, args_copy);
922 int n = vsnprintf(buf ? buf + total_written : nullptr, size > (size_t)total_written ? size - total_written : 0, fmt_token, tmp);
923 va_end(tmp);
924 if (n < 0) {
925 va_end(args_copy);
926 return n;
927 }
928 total_written += n;
929
930 // verbrauchte Argumente aus args_copy ziehen
931 switch (fmt_token[std::strlen(fmt_token) - 1]) {
932 case 'd':
933 case 'i':
934 case 'u':
935 case 'o':
936 case 'x':
937 case 'X': (void)va_arg(args_copy, int); break;
938 case 'f':
939 case 'F':
940 case 'e':
941 case 'E':
942 case 'g':
943 case 'G':
944 case 'a':
945 case 'A': (void)va_arg(args_copy, double); break;
946 case 'c': (void)va_arg(args_copy, int); break;
947 case 'p': (void)va_arg(args_copy, void*); break;
948 case 'n': (void)va_arg(args_copy, int*); break;
949 }
950 }
951 } else {
952 // normales Zeichen kopieren
953 if (total_written < (int)size) {
954 if (buf) buf[total_written] = *p;
955 }
956 total_written++;
957 p++;
958 }
959 }
960
961 // Nullterminierung falls Platz
962 if (buf && total_written < (int)size) {
963 buf[total_written] = '\0';
964 } else if (buf && size > 0) {
965 buf[size - 1] = '\0';
966 }
967
968 va_end(args_copy);
969 return total_written;
970 }
