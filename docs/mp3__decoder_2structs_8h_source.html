43#define D32FP(i, s1, s2) \
44 { \
45 a0 = buf[i]; \
46 a3 = buf[31 - i]; \
47 a1 = buf[15 - i]; \
48 a2 = buf[16 + i]; \
49 b0 = a0 + a3; \
50 b3 = MULSHIFT32(*cptr++, a0 - a3) << 1; \
